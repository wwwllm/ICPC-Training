# 线段树套线段树

## 定义

二维线段树在横坐标的每个线段树结点维护一棵纵坐标线段树，适合静态、稠密矩阵的单点修改与矩形求和。以下实现采用 $0$ 下标和左闭右开区间 `[x1, x2) × [y1, y2)`，支持初始矩阵建树、单点加、单点赋值和矩形和查询。

## 实现

外层结点覆盖一个横坐标区间，内层树维护该区间内各纵坐标区间的元素和。一次单点修改会更新所有经过的外层结点及其内层树；矩形查询将横坐标区间拆成若干外层结点，再在对应内层树中查询纵坐标区间。

### 代码

```cpp
const int inf = 1e9;
struct Info
{
    int max;
    Info() : max(-inf) {};
    Info(int value) : max(value) {};
    Info operator+(const Info &other) const
    {
        Info res;
        res.max = std::max(max, other.max);
        return res;
    }
};
struct SegTree
{
    int n;
    vector<int> Y;
    vector<Info> info;
    SegTree() : n(0) {}
    SegTree(int _n, vector<int> y) : Y(move(y)), n(_n), info(4 * _n) {};
    int findy(int y)
    {
        return ranges::lower_bound(Y, y) - Y.begin();
    }
    void pull(int p) // 更新上移
    {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    // 区间查询[x,y)
    Info query(int p, int l, int r, int x, int y)
    {
        if (l >= y || r <= x)
            return {};
        if (l >= x && r <= y)
        {
            return info[p];
        }
        int mid = l + r >> 1;
        return query(p << 1, l, mid, x, y) + query(p << 1 | 1, mid, r, x, y);
    };
    Info query(int x, int y)
    {
        return query(1, 0, n, x, y);
    };
    // 单点修改
    void modify(int p, int l, int r, int x, const Info &v)
    {
        if (r - l == 1)
        {
            info[p] = info[p] + v;
            return;
        }
        int mid = l + r >> 1;
        if (x < mid)
        {
            modify(p << 1, l, mid, x, v);
        }
        else
        {
            modify(p << 1 | 1, mid, r, x, v);
        }
        pull(p);
    }
    void modify(int x, int v)
    {
        modify(1, 0, n, x, Info(v));
    }
    
};
struct SegTree2D
{
    int n;
    vector<SegTree> info;
    SegTree2D(const vector<vector<int>> &pY)
        : n(pY.size()), info(4 * n)
    {
        build(1, 0, n, pY);
    }
    // 区间查询[x,y)
    Info query(int p, int l, int r, int xl, int xr, int yl, int yr)
    {
        if (l >= xr || r <= xl)
            return {};
        if (l >= xl && r <= xr)
        {
            int L = info[p].findy(yl);
            int R = ranges::upper_bound(info[p].Y, yr) - info[p].Y.begin();
            return info[p].query(L, R);
        }
        int mid = l + r >> 1;
        return query(p << 1, l, mid, xl, xr, yl, yr) + query(p << 1 | 1, mid, r, xl, xr, yl, yr);
    };
    Info query(int xl, int xr, int yl, int yr)
    {
        return query(1, 0, n, xl, xr, yl, yr);
    };
    // 单点修改
    void modify(int p, int l, int r, int x, int y, int v)
    {
        info[p].modify(info[p].findy(y), v);
        if (r - l == 1)
        {
            return;
        }
        int mid = l + r >> 1;
        if (x < mid)
        {
            modify(p << 1, l, mid, x, y, v);
        }
        else
        {
            modify(p << 1 | 1, mid, r, x, y, v);
        }
    }
    void modify(int x, int y, int v)
    {
        modify(1, 0, n, x, y, v);
    }
    void build(int p, int l, int r,
               const vector<vector<int>> &pY)
    {
        if (r - l == 1)
        {
            info[p] = SegTree(pY[l].size(), pY[l]);
            return;
        }

        int mid = (l + r) >> 1;
        build(p << 1, l, mid, pY);
        build(p << 1 | 1, mid, r, pY);

        vector<int> resY;
        merge(info[p << 1].Y.begin(), info[p << 1].Y.end(),
              info[p << 1 | 1].Y.begin(), info[p << 1 | 1].Y.end(),
              back_inserter(resY));

        resY.erase(unique(resY.begin(), resY.end()), resY.end());

        int sz = resY.size();
        info[p] = SegTree(sz, move(resY));
    }
};
```

```cpp
struct BIT2D
{
    int n;
    vector<vector<int>> Y, tr;

    // p：所有可能修改的点 (x, y)
    // x 必须已经离散化为 1 ~ n
    BIT2D(const vector<array<int, 2>> &p, int _n) : n(_n)
    {
        Y.resize(n + 1);
        tr.resize(n + 1);

        // 每个点预先加入它会经过的外层 BIT 节点
        for (auto [x, y] : p)
        {
            for (int i = x; i <= n; i += i & -i)
                Y[i].push_back(y);
        }

        // 每个外层节点单独离散化 y
        for (int i = 1; i <= n; i++)
        {
            ranges::sort(Y[i]);
            Y[i].erase(unique(Y[i].begin(), Y[i].end()), Y[i].end());

            tr[i].assign(Y[i].size() + 1, -inf);
        }
    }

    // 单点取 max：a[x][y] = max(a[x][y], v)
    void modify(int x, int y, int v)
    {
        for (int i = x; i <= n; i += i & -i)
        {
            int j = ranges::lower_bound(Y[i], y) - Y[i].begin() + 1;

            for (; j < tr[i].size(); j += j & -j)
                tr[i][j] = max(tr[i][j], v);
        }
    }

    // 查询所有 x' <= x 且 y' <= y 的最大值
    int query(int x, int y)
    {
        int res = -inf;

        for (int i = x; i > 0; i -= i & -i)
        {
            int j = ranges::upper_bound(Y[i], y) - Y[i].begin();

            for (; j > 0; j -= j & -j)
                res = max(res, tr[i][j]);
        }

        return res;
    }
};
```
### 复杂度

设矩阵大小为 $n \times m$。建树时间为 $O(nm)$；单点加、单点赋值和矩形求和均为 $O(\log n\log m)$；空间复杂度为 $O(nm)$，常数较大。
