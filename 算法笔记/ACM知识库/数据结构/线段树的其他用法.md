# 线段树的其他用法

## 模块名称

线段树的其他用法

## 来源

- 数据结构/线段树的其他用法.md

## 适用场景

区间加、区间查询、单点修改、动态维护聚合信息。适合题面出现“区间”“修改”“查询”“最值/和”等关键词。

## 核心思想

把整体区间递归拆成若干线段节点，每个节点保存可合并的信息；懒标记表示对子区间暂缓执行的修改。

# 前缀和的最大值
```cpp
const int inf = 4e18;
struct Info
{
    int sum, max;
    Info() : sum(), max(-inf) {};
    Info(int val, int max = -inf) : sum(val), max(max) {};
    Info operator+(const Info &other) const
    {
        Info res;
        res.sum = sum + other.sum;
        res.max = std::max(max, sum + other.max);
        return res;
    }
};
struct SegTree
{
    int n;
    vector<Info> info;
    SegTree(int _n) : n(_n), info(4 * _n) {};
    void pull(int p) // 更新上移
    {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    // 区间查询[x,y)
    Info query(int p, int l, int r, int x, int y)
    {
        if (l >= y || r <= x)
            return {};
        if (l >= x && r <= y)
        {
            return info[p];
        }
        int mid = l + r >> 1;
        return query(p << 1, l, mid, x, y) + query(p << 1 | 1, mid, r, x, y);
    };
    Info query(int x, int y)
    {
        return query(1, 0, n, x, y);
    };
    // 单点修改
    void modify(int p, int l, int r, int x, const Info &v)
    {
        if (r - l == 1)
        {
            info[p] = v;
            return;
        }
        int mid = l + r >> 1;
        if (x < mid)
        {
            modify(p << 1, l, mid, x, v);
        }
        else
        {
            modify(p << 1 | 1, mid, r, x, v);
        }
        pull(p);
    }
    void modify(int x, Info v)
    {
        modify(1, 0, n, x, v);
    }
};
```
# 前缀最大值的和
```cpp
const int inf = 1e18;
struct Info
{
    int sum, max;
    Info() : sum(), max(-inf) {};
    Info(int value) : sum(value), max(value) {};
};
struct SegTree
{
    int n;
    vector<Info> info;
    SegTree(int _n) : n(_n), info(4 * _n) {};
    int calc(int p, int l, int r, int val)
    {
        if (r - l == 1)
            return std::max(val, info[p].max);
        int mid = l + r >> 1;
        if (val >= info[p << 1].max)
        {
            int lres = val * (mid - l);
            return lres + calc(p << 1 | 1, mid, r, val);
        }
        else
        {
            int lres = calc(p << 1, l, mid, val);
            int rres = info[p].sum - info[p << 1].sum;
            return lres + rres;
        }
    }
    void pull(int p, int l, int r) // 更新上移
    {
        info[p].max = std::max(info[p << 1].max, info[p << 1 | 1].max);
        info[p].sum = info[p << 1].sum + calc(p << 1 | 1, l + r >> 1, r, info[p << 1].max);
    }
    // 区间查询[x,y)
    int query(int p, int l, int r, int x, int y, int &cur)
    {
        if (l >= y || r <= x)
            return {};
        if (l >= x && r <= y)
        {
            int res = calc(p, l, r, cur);
            cur = max(info[p].max, cur);
            return res;
        }
        int res = 0;
        int mid = l + r >> 1;
        if (x < mid)
            res += query(p << 1, l, mid, x, y, cur);

        if (y > mid)
            res += query(p << 1 | 1, mid, r, x, y, cur);

        return res;
    };
    int query(int x, int y)
    {
        int cur = -inf;
        return query(1, 0, n, x, y, cur);
    };
    // 单点修改
    void modify(int p, int l, int r, int x, const Info &v)
    {
        if (r - l == 1)
        {
            info[p] = v;
            return;
        }
        int mid = l + r >> 1;
        if (x < mid)
        {
            modify(p << 1, l, mid, x, v);
        }
        else
        {
            modify(p << 1 | 1, mid, r, x, v);
        }
        pull(p, l, r);
    }
    void modify(int x, int v)
    {
        modify(1, 0, n, x, Info(v));
    }
};
```

## 复杂度

建树 O(n)，单次查询/修改 O(log n)，空间 O(n)。

## 常见错误

- `mid = l + r >> 1` 建议写成 `(l + r) >> 1`，避免读代码时误判优先级。
- 区间统一使用 `[l, r)` 或 `[l, r]`，不要混用。
- 下放懒标记前要保证当前节点确实有标记。
