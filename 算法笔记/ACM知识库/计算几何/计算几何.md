# 计算几何

## 模块名称

计算几何

## 来源

- 计算几何.md

## 适用场景

点、向量、直线、线段、圆相关的几何判定和计算。

## 核心思想

用点积处理投影/夹角，用叉积处理方向/面积/相交，用 eps 统一浮点比较。

```cpp
#include <bits/stdc++.h>
using namespace std;

const double eps = 1e-9;
const double pi = acos(-1.0);

using T = double;
int sign(T x)
{
    // x为int改成x==0
    if (fabs(x) < eps)
        return 0;
    return x < 0 ? -1 : 1;
}

int cmp(T x, T y)
{
    return sign(x - y);
}

double Acos(double x)
{
    if (x < -1)
        return pi;
    if (x > 1)
        return 0;
    return acos(x);
}

// ==========================================
// 点与向量的结构体
// ==========================================
struct Point;
using Vector = Point;

struct Point
{
    T x, y;
    Point(T x = 0, T y = 0) : x(x), y(y) {}

    Point operator+(const Point &b) const { return Point(x + b.x, y + b.y); }
    Point operator-(const Point &b) const { return Point(x - b.x, y - b.y); }
    Point operator*(T k) const { return Point(x * k, y * k); }
    Point operator/(T k) const { return Point(x / k, y / k); }

    friend istream &operator>>(istream &in, Point &p)
    {
        in >> p.x >> p.y;
        return in;
    }

    bool operator==(const Point &b) const
    {
        return cmp(x, b.x) == 0 && cmp(y, b.y) == 0;
    }

    // 点的大小比较：以 x 为第一关键字，y 为第二关键字（常用于排序）
    bool operator<(const Point &b) const
    {
        int c = cmp(x, b.x);
        if (c != 0)
            return c == -1;
        return cmp(y, b.y) == -1;
    }

    // 计算向量模长（两点间距离）
    double len() const
    {
        return hypot(x, y);
    }

    // 计算向量模长的平方
    T len2() const
    {
        return x * x + y * y;
    }

    // 求当前向量的法向量（逆时针旋转 90 度）
    Vector normal() const
    {
        return Point(-y, x);
    }

    // 求该点/向量关于原点的极角，范围在 [-pi, pi] 之间
    double p_angle() const
    {
        return atan2(y, x);
    }

    // 向量单位化（归一化）
    Vector unit() const
    {
        double l = this->len();
        if (sign(l) == 0)
            return (*this);
        return (*this) / l;
    }

    // 向量逆时针旋转给定弧度 angle
    Vector rotate(double angle) const
    {
        double c = cos(angle), s = sin(angle);
        return Point(x * c - y * s, x * s + y * c);
    }
};

// 计算两个向量的点积（内积）
T dot(const Vector &a, const Vector &b)
{
    return a.x * b.x + a.y * b.y;
}

// 计算两个向量的叉积（外积）
T cross(const Vector &a, const Vector &b)
{
    return a.x * b.y - b.x * a.y;
}

// 计算两个向量的夹角，范围在 [0, pi] 之间
double get_angle(const Vector &a, const Vector &b)
{
    return fabs(atan2(cross(a, b), dot(a, b)));
}

// 计算以点 a 为起点，ab 和 ac 构成的平行四边形的有向面积
T area(const Point &a, const Point &b, const Point &c)
{
    return cross(b - a, c - a);
}

// 判断三点是否共线
bool is_collinear(const Point &a, const Point &b, const Point &c)
{
    return sign(cross(b - a, c - a)) == 0;
}

// 求当前点关于点 o 的对称点
Point reflect(const Point &p, const Point &o)
{
    return o * 2 - p;
}

// 精准全平面极角排序比较器（按逆时针排序，不使用 atan2，防精度误差）
struct Polarcmp
{
    // 象限判断：1->右上(含X正), 2->左上(含Y正), 3->左下(含X负), 4->右下(含Y负)
    int quad(const Point &p) const
    {
        if (p.x >= 0 && p.y >= 0)
            return 1;
        if (p.x < 0 && p.y >= 0)
            return 2;
        if (p.x < 0 && p.y < 0)
            return 3;
        return 4;
    }

    // 比较函数
    bool operator()(const Point &a, const Point &b) const
    {
        int qa = quad(a), qb = quad(b);
        if (qa != qb)
            return qa < qb; // 象限不同，象限小的排前面
        T c = cross(a, b);
        if (sign(c) != 0)
            return sign(c) > 0;     // 象限相同，叉积大于0说明 b 在 a 的左侧（逆时针）
        return a.len2() < b.len2(); // 极角相同时，距离近的排前面
    }
};

// ==========================================
// 直线
// ==========================================
struct Line
{
    Point p;  // 直线上的基准点
    Vector v; // 直线的方向向量（参数方程：p + t*v）
    Line() {}
    Line(Point p, Vector v) : p(p), v(v) {}

    // 判定点 a 与有向直线（p -> p+v） Orient/To-Left 的位置关系
    // 返回值：1 在直线左侧 | -1 在直线右侧 | 0 在直线上
    int to_left(const Point &a) const
    {
        return sign(cross(v, a - p));
    }

    // 计算点 a 到该直线的垂直距离
    double dist(const Point &a) const
    {
        return fabs(cross(v, a - p)) / v.len();
    }

    // 求点 a 在该直线上的垂直投影点
    Point proj(const Point &a) const
    {
        return p + v * (dot(a - p, v) / v.len2());
    }

    // 求点 a 关于该直线的对称点（镜像点）
    Point reflect(const Point &a) const
    {
        Point pr = proj(a);
        return pr * 2.0 - a;
    }
};
// 判定两条直线的关系
// 返回值：0 相交 | 1 平行 | 2 重合
static int line_relation(const Line &a, const Line &b)
{
    if (sign(cross(a.v, b.v)) != 0)
        return 0;
    if (sign(cross(a.v, b.p - a.p)) == 0)
        return 2;
    return 1;
}

// 求两条直线 a 和 b 的唯一交点（需提前断定它们不平行）
Point inter_l2l(const Line &a, const Line &b)
{
    assert(sign(cross(a.v, b.v)) != 0);
    Vector u = a.p - b.p;
    double t = cross(b.v, u) / cross(a.v, b.v);
    return a.p + a.v * t;
}

// ==========================================
// 线段
// ==========================================
struct Segment
{
    Point a, b; // 线段的两个端点
    Segment() {}
    Segment(Point a, Point b) : a(a), b(b) {}

    // 判断点 p 与当前线段的位置关系：-1 在端点上 | 1 严格在线段内部 | 0 不在线段上
    int on_segment(const Point &p) const
    {
        if (p == a || p == b)
            return -1;
        if (sign(cross(p - a, p - b)) == 0 && sign(dot(p - a, p - b)) <= 0)
            return 1;
        return 0;
    }

    // 判断两线段是否相交：-1 端点处接触相交 | 1 严格在内部交叉相交 | 0 互不相交
    int is_inter(const Segment &s) const
    {
        if (on_segment(s.a) != 0 || on_segment(s.b) != 0 || s.on_segment(a) != 0 || s.on_segment(b) != 0)
            return -1;
        T c1 = cross(b - a, s.a - a), c2 = cross(b - a, s.b - a);
        T c3 = cross(s.b - s.a, a - s.a), c4 = cross(s.b - s.a, b - s.a);
        return (sign(c1) * sign(c2) < 0 && sign(c3) * sign(c4) < 0) ? 1 : 0;
    }

    // 计算点 p 到当前线段的最短距离
    double dist(const Point &p) const
    {
        if (a == b)
            return (p - a).len();
        Vector v1 = b - a, v2 = p - a, v3 = p - b;
        if (sign(dot(v1, v2)) < 0)
            return v2.len();
        if (sign(dot(v1, v3)) > 0)
            return v3.len();
        Line l{a, b - a};
        return l.dist(p); // 正常向线段作垂线距离
    }

    // 计算当前线段到另一条线段 s 的最短距离
    double dist(const Segment &s) const
    {
        if (is_inter(s) != 0)
            return 0;
        return min({dist(s.a), dist(s.b), s.dist(a), s.dist(b)});
    }
};

// 求两条线段的交点坐标（调用前必须先确保 is_inter 返回非 0 状态）
Point inter_s2s(const Segment &s1, const Segment &s2)
{
    assert(s1.is_inter(s2) == 0);
    Line l1{s1.a, s1.b - s1.a};
    Line l2{s2.a, s2.b - s2.a};
    return inter_l2l(l1, l2);
}

struct Circle
{
    Point o;
    T r;
    Circle() {};
    Circle(Point o, T r) : o(o), r(r) {};

    bool operator==(const Circle &a) const
    {
        return o == a.o && cmp(r, a.r) == 0;
    }
};

// ==========================================
// 多边形
// ==========================================
struct Polygon
{
    vector<Point> p; // 顶点按边界顺序存储，顺/逆时针均可
    Polygon() {}
    Polygon(const vector<Point> &p) : p(p) {}

    int size() const
    {
        return p.size();
    }

    // 有向面积的两倍：逆时针为正，顺时针为负
    T area2() const
    {
        T res = 0;
        for (int i = 0; i < size(); i++)
            res += cross(p[i], p[(i + 1) % size()]);
        return res;
    }

    // 多边形面积
    double area() const
    {
        return fabs(area2()) / 2;
    }

    // 多边形周长
    double perimeter() const
    {
        double res = 0;
        for (int i = 0; i < size(); i++)
            res += (p[i] - p[(i + 1) % size()]).len();
        return res;
    }

    // 判断点 a 与多边形的位置关系：1 内部 | -1 边界上 | 0 外部
    int contain(const Point &a) const
    {
        bool in = false;
        for (int i = 0; i < size(); i++)
        {
            Point u = p[i], v = p[(i + 1) % size()];
            if (Segment(u, v).on_segment(a) != 0)
                return -1;
            if (sign(u.y - a.y) != sign(v.y - a.y) &&
                sign(cross(v - u, a - u)) == (sign(v.y - a.y) > 0 ? 1 : -1))
                in = !in;
        }
        return in ? 1 : 0;
    }

    // 判断是否为凸多边形，退化情形返回 false
    bool is_convex() const
    {
        if (size() < 3 || sign(area2()) == 0)
            return false;
        int pre = 0;
        for (int i = 0; i < size(); i++)
        {
            int cur = sign(cross(p[(i + 1) % size()] - p[i], p[(i + 2) % size()] - p[(i + 1) % size()]));
            if (cur == 0)
                continue;
            if (pre != 0 && cur != pre)
                return false;
            pre = cur;
        }
        return pre != 0;
    }
};

// ==========================================
// 凸包
// ==========================================
struct Convex : Polygon
{
    Convex() {}
    Convex(vector<Point> q, bool keep_collinear = false)
    {
        build(q, keep_collinear);
    }

    // Andrew 单调链：默认删除边上的共线点；keep_collinear 为 true 时保留
    void build(vector<Point> q, bool keep_collinear = false)
    {
        sort(q.begin(), q.end());
        q.erase(unique(q.begin(), q.end()), q.end());
        p.clear();
        if (q.size() <= 1)
        {
            p = q;
            return;
        }
        bool collinear = true;
        for (int i = 2; i < (int)q.size(); i++)
            if (sign(cross(q[1] - q[0], q[i] - q[0])) != 0)
            {
                collinear = false;
                break;
            }
        if (collinear)
        {
            p = keep_collinear ? q : vector<Point>{q.front(), q.back()};
            return;
        }

        auto check = [&](const Point &a, const Point &b, const Point &c)
        {
            int s = sign(cross(b - a, c - b));
            return keep_collinear ? s < 0 : s <= 0;
        };

        for (const Point &a : q)
        {
            while (p.size() >= 2 && check(p[p.size() - 2], p.back(), a))
                p.pop_back();
            p.push_back(a);
        }
        int lower_size = p.size();
        for (int i = (int)q.size() - 2; i >= 0; i--)
        {
            while ((int)p.size() > lower_size && check(p[p.size() - 2], p.back(), q[i]))
                p.pop_back();
            p.push_back(q[i]);
        }
        p.pop_back(); // 不重复存储起点
    }
    // 旋转卡壳求凸包直径
    double diameter()
    {
        int n = p.size();
        if (n <= 1)
            return 0;
        if (n == 2)
            return (p[0] - p[1]).len();
        double ans = 0;
        for (int i = 0, j = i + 1; i < n; i++)
        {
            auto u = p[i], v = p[(i + 1) % n];
            while (cmp(abs(cross(v - u, p[(j + 1) % n] - u)), abs(cross(v - u, p[j] - u))) == 1)
            {
                j = (j + 1) % n;
            }
            ans = max(ans, (v - p[j]).len());
            ans = max(ans, (u - p[j]).len());
        }
        return ans;
    }
};

// 去除同方向半平面
vector<Line> uniquel(vector<Line> &l)
{
    vector<Line> ll;
    for (auto line : l)
    {
        if (ll.empty() || sign(cross(ll.back().v, line.v)) != 0)
            ll.push_back(line);
        else
        {
            if (ll.back().to_left(line.p) == 1)
            {
                ll.back() = line;
            }
        }
    }
    return ll;
}

Convex inter_con(vector<Line> &line)
{
    ranges::sort(line, [](auto &a, auto &b)
                 { return Polarcmp()(a.v, b.v); });
    line = uniquel(line);
    deque<Line> dq;
    for (auto &l : line)
    {
        while (dq.size() > 1)
        {
            Point p = inter_l2l(dq[dq.size() - 2], dq.back());
            // 0：边界，1：左侧
            if (l.to_left(p) >= 0)
                break;
            dq.pop_back();
        }
        while (dq.size() > 1)
        {
            Point p = inter_l2l(dq[0], dq[1]);

            if (l.to_left(p) >= 0)
                break;

            dq.pop_front();
        }
        dq.push_back(l);
    }
    while (dq.size() > 2)
    {
        Point p = inter_l2l(dq[dq.size() - 2], dq.back());
        if (dq[0].to_left(p) >= 0)
            break;
        dq.pop_back();
    }
    while (dq.size() > 2)
    {
        Point p = inter_l2l(dq[0], dq[1]);
        if (dq.back().to_left(p) >= 0)
            break;
        dq.pop_front();
    }
    vector<Point> res;
    if (dq.size() < 3)
        return res;
    int n = dq.size();
    for (int i = 0; i < n; i++)
    {
        res.push_back(inter_l2l(dq[i], dq[(i + 1) % n]));
    }
    return Convex(res);
}

// ==========================================
// 闵可夫斯基和
// 输入：两个逆时针凸多边形
// 返回：A+B 的凸多边形
// ==========================================

Convex minkowski(Convex &a, Convex &b)
{
    int n = a.size(), m = b.size();

    // 找最低点
    auto get = [](vector<Point> &p)
    {
        int pos = 0;

        for (int i = 1; i < p.size(); i++)
        {
            if (p[i].y < p[pos].y ||
                (cmp(p[i].y, p[pos].y) == 0 &&
                 p[i].x < p[pos].x))
            {
                pos = i;
            }
        }

        rotate(p.begin(), p.begin() + pos, p.end());
    };
    auto &pa = a.p, &pb = b.p;
    get(pa), get(pb);

    vector<Vector> va(n), vb(m);
    for (int i = 0; i < n; i++)
    {
        va[i] = pa[(i + 1) % n] - pa[i];
    }

    for (int i = 0; i < m; i++)
    {
        vb[i] = pb[(i + 1) % m] - pb[i];
    }

    vector<Point> res;

    int i = 0, j = 0;

    Point cur = pa[0] + pb[0];

    while (i != n || j != m)
    {
        Vector v;
        if (i == n)
            v = vb[j++];
        else if (j == m)
            v = va[i++];
        else
        {
            int c = sign(cross(va[i], vb[j]));

            if (c > 0)
            {
                v = va[i++];
            }
            else if (c < 0)
            {
                v = vb[j++];
            }
            else
            {
                v = va[i++] + vb[j++];
            }
        }
        cur = cur + v;
        res.push_back(cur);
    }
    res.pop_back();
    return Convex(res);
}
```

## 常见错误

- 不要直接比较 double 相等。
- 叉积方向和端点接触返回值要先约定。
- 断言条件要和函数返回值一致。
